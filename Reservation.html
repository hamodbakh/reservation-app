<script>

    let reservations = JSON.parse(localStorage.getItem('restaurant_reservations')) || [];


    const form = document.getElementById('reservationForm');

    const tableBody = document.getElementById('reservationTableBody');

    const searchInput = document.getElementById('searchInput');

    const emptyState = document.getElementById('emptyState');

    const editIndexInput = document.getElementById('editIndex');

    const formTitle = document.getElementById('formTitle');

    const submitBtn = document.getElementById('submitBtn');

    const cancelEditBtn = document.getElementById('cancelEditBtn');


    document.getElementById('date').valueAsDate = new Date();


    form.addEventListener('submit', function(e) {

        e.preventDefault();


        const reservationData = {

            id: editIndexInput.value === "" ? Date.now() : Number(editIndexInput.value),

            name: document.getElementById('customerName').value,

            phone: document.getElementById('phone').value,

            guests: document.getElementById('guests').value,

            table: document.getElementById('tableNumber').value,

            date: document.getElementById('date').value,

            time: document.getElementById('time').value,

            occasion: document.getElementById('occasion').value,

            notes: document.getElementById('notes').value,

            status: 'Confirmed'

        };


        const editIdx = editIndexInput.value;

        if (editIdx === "") {

            reservations.push(reservationData);

        } else {

            const index = reservations.findIndex(r => r.id == editIdx);

            if (index !== -1) {

                reservations[index] = reservationData;

            }

            resetForm();

        }


        saveAndRender();

        form.reset();

        document.getElementById('date').valueAsDate = new Date();

    });


    window.editReservation = function(id) {

        const res = reservations.find(r => r.id == id);

        if (!res) return;


        document.getElementById('customerName').value = res.name;

        document.getElementById('phone').value = res.phone;

        document.getElementById('guests').value = res.guests;

        document.getElementById('tableNumber').value = res.table;

        document.getElementById('date').value = res.date;

        document.getElementById('time').value = res.time;

        document.getElementById('occasion').value = res.occasion || 'None';

        document.getElementById('notes').value = res.notes || '';

        editIndexInput.value = res.id;


        formTitle.textContent = "Edit Reservation";

        submitBtn.textContent = "Update Reservation";

        cancelEditBtn.style.display = "block";

        window.scrollTo({ top: 0, behavior: 'smooth' });

    }


    window.resetForm = function() {

        form.reset();

        editIndexInput.value = "";

        formTitle.textContent = "New Reservation";

        submitBtn.textContent = "Save & Confirm Reservation";

        cancelEditBtn.style.display = "none";

        document.getElementById('date').valueAsDate = new Date();

    }


    window.deleteReservation = function(id) {

        if(confirm('Are you sure you want to cancel this reservation?')) {

            reservations = reservations.filter(res => res.id !== id);

            saveAndRender();

        }

    }


    searchInput.addEventListener('input', function() {

        renderTable(this.value.toLowerCase());

    });


    function saveAndRender() {

        localStorage.setItem('restaurant_reservations', JSON.stringify(reservations));

        renderTable(searchInput.value.toLowerCase());

    }


    function getOccasionIcon(occasion) {

        switch(occasion) {

            case 'Birthday': return '🎂 ';

            case 'Anniversary': return '💍 ';

            case 'VIP': return '⭐ ';

            case 'Other': return '🎉 ';

            default: return '';

        }

    }


    function renderTable(filter = '') {

        tableBody.innerHTML = '';

        

        const filtered = reservations.filter(res => 

            res.name.toLowerCase().includes(filter) || 

            res.phone.toLowerCase().includes(filter) ||

            (res.notes && res.notes.toLowerCase().includes(filter))

        );


        filtered.sort((a, b) => new Date(`${a.date}T${a.time}`) - new Date(`${b.date}T${b.time}`));


        if (filtered.length === 0) {

            emptyState.style.display = 'block';

            return;

        } else {

            emptyState.style.display = 'none';

        }


        filtered.forEach(res => {

            let cleanPhone = res.phone.replace(/[^0-9+]/g, '');

            let icon = getOccasionIcon(res.occasion);

            let occasionText = res.occasion && res.occasion !== 'None' ? `[${res.occasion}] ` : '';

            let notesText = res.notes ? ` - Note: ${res.notes}` : '';

            

            let msg = `Hello ${res.name}, this is a reminder from our restaurant for your table reservation (${res.table}) on ${res.date} at ${res.time} for ${res.guests} guests.${occasionText}${notesText} We look forward to seeing you!`;

            let encodedMsg = encodeURIComponent(msg);

            let waLink = `https://wa.me/${cleanPhone}?text=${encodedMsg}`;


            const row = document.createElement('tr');

            row.innerHTML = `

                <td>

                    <strong>${icon}${escapeHtml(res.name)}</strong>

                    ${res.occasion && res.occasion !== 'None' ? `<br><small class="badge" style="background:#fef3c7; color:#92400e;">${res.occasion}</small>` : ''}

                    ${res.notes ? `<div class="note-text">📝 ${escapeHtml(res.notes)}</div>` : ''}

                </td>

                <td>${escapeHtml(res.phone)}</td>

                <td>${res.guests}</td>

                <td>${escapeHtml(res.table)}</td>

                <td>${res.date} <br><small style="color:#64748b">${res.time}</small></td>

                <td><span class="badge badge-confirmed">${res.status}</span></td>

                <td>

                    <div class="action-btns">

                        <a href="${waLink}" target="_blank" class="btn-wa">Remind</a>

                        <button class="btn-edit" onclick="editReservation(${res.id})">Edit</button>

                        <button class="btn-delete" onclick="deleteReservation(${res.id})">Cancel</button>

                    </div>

                </td>

            `;

            tableBody.appendChild(row);

        });

    }


    function escapeHtml(str) {

        return str.replace(/[&<>'"]/g, 

            tag => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', "'": '&#39;', '"': '&quot;' }[tag] || tag)

        );

    }


    renderTable();

</script>
